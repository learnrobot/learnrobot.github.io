<script>
    // --- Navigation & Scroll ---
    function scrollToId(id) {
        const el = document.getElementById(id);
        if(el) el.scrollIntoView({ behavior: 'smooth' });
    }

    const mainArea = document.getElementById("main-scroll-area");
    const topBtn = document.getElementById("topBtn");
    
    if (mainArea && topBtn) {
        mainArea.addEventListener('scroll', () => {
            topBtn.style.display = mainArea.scrollTop > 300 ? "flex" : "none";
        });
    }

    function scrollToTop() {
        if(mainArea) mainArea.scrollTo({ top: 0, behavior: 'smooth' });
    }

    // --- Interactive Labs Logic ---
    
    // Balance Lab
    function equipIMU() {
        const bot = document.getElementById('balance-bot');
        const msg = document.getElementById('balance-msg');
        bot.style.transform = 'rotate(0deg)';
        bot.style.color = '#2ecc71';
        msg.innerHTML = 'IMU Sensor Online! Balance restored.';
        msg.style.color = '#2ecc71';
    }
    function removeIMU() {
        const bot = document.getElementById('balance-bot');
        const msg = document.getElementById('balance-msg');
        bot.style.transform = 'rotate(45deg)';
        bot.style.color = '#FF5555';
        msg.innerHTML = 'IMU Sensor Missing!';
        msg.style.color = '#FF5555';
    }

    // Battery Lab
    let batt = 100;
    function useBattery(amt) {
        batt = Math.max(0, batt - amt);
        updateBatt();
    }
    function recharge() {
        batt = 100;
        updateBatt();
    }
    function updateBatt() {
        const bar = document.getElementById('battery-level');
        bar.style.width = batt + '%';
        bar.innerText = batt + '%';
        bar.style.background = batt > 40 ? '#2ecc71' : (batt > 15 ? '#f1c40f' : '#e74c3c');
    }

    // Path Planning Lab
    let obsActive = false;
    function toggleObstacle() {
        obsActive = !obsActive;
        const obs = document.getElementById('maze-obs');
        const bot = document.getElementById('maze-bot');
        const msg = document.getElementById('maze-msg');
        
        if (obsActive) {
            obs.style.display = 'block';
            bot.style.left = '30%';
            msg.innerHTML = '<span style="color:#FF5555">Obstacle detected!</span> Detouring...';
        } else {
            obs.style.display = 'none';
            bot.style.left = '80%';
            msg.innerHTML = 'Path clear. Goal reached.';
        }
    }

    // Vision Lab
    let visionActive = false;
    function toggleVisionLab() {
        visionActive = !visionActive;
        const box = document.getElementById('bounding-box');
        const text = document.getElementById('vision-text');
        
        if (visionActive) {
            box.style.display = 'flex';
            box.style.justifyContent = 'center';
            box.style.alignItems = 'center';
            text.style.display = 'none';
        } else {
            box.style.display = 'none';
            text.style.display = 'block';
        }
    }

    // Sidebar Dynamic Spotlight Rotation
    const players = [
        { name: 'Boston Dynamics', cat: 'Humanoid & Quadruped', url: 'https://bostondynamics.com' },
        { name: 'FANUC', cat: 'Industrial Automation', url: 'https://www.fanuc.com' },
        { name: 'Universal Robots', cat: 'Collaborative Robots', url: 'https://www.universal-robots.com' }
    ];
    let pIdx = 0;
    setInterval(() => {
        pIdx = (pIdx + 1) % players.length;
        document.getElementById('sl-name').innerText = players[pIdx].name;
        document.getElementById('sl-category').innerText = players[pIdx].cat;
        document.getElementById('sl-link').href = players[pIdx].url;
    }, 5000);

</script>
